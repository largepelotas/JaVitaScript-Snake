# One build description for both targets (PLAN.md 7.1), selected by whether
# CMAKE_TOOLCHAIN_FILE points at vita.toolchain.cmake - which sets VITA.
#
#   cmake -B build-host && cmake --build build-host
#   cmake -B build-vita -DCMAKE_TOOLCHAIN_FILE=$VITASDK/share/vita.toolchain.cmake
#   cmake --build build-vita
#
# The Makefile beside this file is a separate, faster loop for the host tests
# and the headless screenshots; it is not the shipping build.
cmake_minimum_required(VERSION 3.16)

project(vita-snake C)

set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

add_compile_options(
  -Wall -Wextra -Wpedantic -Wshadow -Wconversion -Wstrict-prototypes
)

# The version, and the only place it is written down. Three consumers want
# three different shapes of it - the startup log wants all of it, param.sfo
# takes two fields, release-vita.yml checks it against the tag - so it is
# derived here rather than transcribed into each of them and left to drift.
#
# Leading zeros are rejected so that "01.2.3" cannot mean two things; the CI
# check applies the same grammar to the tag.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" SNAKE_VERSION)
string(STRIP "${SNAKE_VERSION}" SNAKE_VERSION)

if(NOT SNAKE_VERSION MATCHES "^(0|[1-9][0-9]*)\\.(0|[1-9][0-9]*)\\.(0|[1-9][0-9]*)$")
  message(FATAL_ERROR
          "VERSION must hold MAJOR.MINOR.PATCH, not '${SNAKE_VERSION}'")
endif()

set(V_MAJOR ${CMAKE_MATCH_1})
set(V_MINOR ${CMAKE_MATCH_2})

# param.sfo holds XX.YY: major and minor, two zero-padded digits each. That is
# why v1.0.0 shipped as "01.00" and v1.1.0 as "01.01" - "01.10" would be a
# v1.10.0. The patch component has nowhere to go, so a v1.1.1 reads "01.01"
# too, and the startup log is what tells the two apart on hardware.
if(V_MAJOR GREATER 99 OR V_MINOR GREATER 99)
  message(FATAL_ERROR
          "param.sfo holds two digits per field, so ${SNAKE_VERSION} cannot "
          "be expressed")
endif()

if(V_MAJOR LESS 10)
  set(V_MAJOR "0${V_MAJOR}")
endif()
if(V_MINOR LESS 10)
  set(V_MINOR "0${V_MINOR}")
endif()

set(VITA_VERSION "${V_MAJOR}.${V_MINOR}")
message(STATUS "snake ${SNAKE_VERSION} (param.sfo APP_VER ${VITA_VERSION})")

# The core and the shell are byte-identical on both targets; only the platform
# file differs (PLAN.md 1.2). This list is the whole layering rule, mechanised.
file(GLOB CORE_SOURCES  ${CMAKE_CURRENT_SOURCE_DIR}/src/core/*.c)
file(GLOB SHELL_SOURCES ${CMAKE_CURRENT_SOURCE_DIR}/src/shell/*.c)
set(COMMON_SOURCES ${CMAKE_CURRENT_SOURCE_DIR}/src/main.c
                   ${CORE_SOURCES} ${SHELL_SOURCES})

if(VITA)
  include("${VITASDK}/share/vita.cmake" REQUIRED)

  # param.sfo's TITLE: the label under the LiveArea bubble and in Content
  # Manager. Renamed from "Snake" in v1.1.2.
  #
  # Note what this is NOT tied to. VITA_TITLEID below and the ux0:data path in
  # platform_vita.c are both independent of it, and both must stay put: a new
  # title ID makes the firmware treat this as a different application - second
  # bubble, the old one left behind - and a renamed data directory strands
  # every existing highscore and theme. A rename is a label change and nothing
  # more.
  set(VITA_APP_NAME "JaVitaScript Snake")
  # Checked against all 1022 VitaDB homebrew entries and unused.
  # Changing it means touching this line and the ux0:app path in DEPLOY notes.
  set(VITA_TITLEID  "SNEK00001")

  add_executable(snake
    ${COMMON_SOURCES}
    ${CMAKE_CURRENT_SOURCE_DIR}/src/platform/platform_vita.c
  )

  target_compile_definitions(snake PRIVATE SNAKE_VERSION="${SNAKE_VERSION}")

  # Debug info in the unstripped ELF, which is what vita-parse-core reads a
  # crash dump against (PLAN.md 7.5). Without it a dump symbolises to function
  # names but "??:?" for the source line, which is half a stack trace. This
  # costs nothing on device: eboot.bin is built from the stripped SELF, so the
  # shipped binary is unchanged - only build-vita/snake grows.
  target_compile_options(snake PRIVATE -g)

  # find_package, not pkg-config: 102 of the 107 .pc files vdpm ships hardcode
  # prefix=/usr/local/vitasdk and SDL2_ttf.pc emits a malformed -L at every
  # prefix (PLAN.md 2.5). The CMake configs are correct.
  find_package(SDL2 REQUIRED)
  find_package(SDL2_ttf REQUIRED)

  # SDL2_ttf's config names Freetype but not what Freetype itself was built
  # against, and static linking will not chase that for us: without these the
  # link fails on png_*, BZ2_* and inflate*. Order matters - dependencies come
  # after the libraries that need them (PLAN.md 2.5 makes the same point about
  # the SDL2 link list).
  target_link_libraries(snake
    SDL2_ttf::SDL2_ttf-static
    SDL2::SDL2
    freetype png16 bz2 z m
  )

  vita_create_self(snake.self snake UNSAFE)

  # LiveArea assets (PLAN.md 7.3). These are checked in, generated by
  # tools/gen_livearea.c (`make livearea`) from the game's own palette, and
  # template.xml is the VitaSDK hello_world one unchanged. Without the FILE
  # lines the vpk still installs, but with Sony's default placeholder bubble.
  vita_create_vpk(snake.vpk ${VITA_TITLEID} snake.self
    VERSION ${VITA_VERSION}
    NAME ${VITA_APP_NAME}
    FILE ${CMAKE_CURRENT_SOURCE_DIR}/sce_sys/icon0.png sce_sys/icon0.png
    FILE ${CMAKE_CURRENT_SOURCE_DIR}/sce_sys/livearea/contents/bg.png
         sce_sys/livearea/contents/bg.png
    FILE ${CMAKE_CURRENT_SOURCE_DIR}/sce_sys/livearea/contents/startup.png
         sce_sys/livearea/contents/startup.png
    FILE ${CMAKE_CURRENT_SOURCE_DIR}/sce_sys/livearea/contents/template.xml
         sce_sys/livearea/contents/template.xml
    FILE ${CMAKE_CURRENT_SOURCE_DIR}/assets/font.ttf assets/font.ttf
    FILE ${CMAKE_CURRENT_SOURCE_DIR}/assets/font-LICENSE.txt assets/font-LICENSE.txt
  )
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
  pkg_check_modules(SDL2TTF REQUIRED IMPORTED_TARGET SDL2_ttf)

  add_executable(snake
    ${COMMON_SOURCES}
    ${CMAKE_CURRENT_SOURCE_DIR}/src/platform/platform_desktop.c
  )
  target_link_libraries(snake PkgConfig::SDL2TTF PkgConfig::SDL2)
  target_compile_definitions(snake PRIVATE SNAKE_VERSION="${SNAKE_VERSION}")

  # The desktop binary looks for assets/ next to itself as well as in the
  # working directory (see plat_asset_path), so a build-tree run works without
  # an install step.
  add_custom_command(TARGET snake POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory
            ${CMAKE_CURRENT_SOURCE_DIR}/assets $<TARGET_FILE_DIR:snake>/assets
    VERBATIM
  )
endif()
